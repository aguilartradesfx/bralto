# Plan Q4 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Sección `/plan-q4` del panel admin donde el equipo navega el plan del trimestre por pestañas y trabaja sus tareas (marcar, reasignar, fechas, notas, semanales).

**Architecture:** Lógica pura en `lib/plan-q4/` (fechas CR, parser del archivo del plan, reglas, validación, diff del seed) con tests `node --test`. El contenido y las tareas viven en Supabase (`plan_content`, `plan_team`, `plan_tasks`, `plan_task_weeks`, RLS sin políticas); un script de seed los carga desde `plan-q4-admin.md` (fuera de git). La página es un server component con piezas cliente para editar; las escrituras pasan por 3 APIs con `requireApiPermission('can_view_plan')`.

**Tech Stack:** Next.js 15 App Router, TypeScript, Tailwind 3.4, Supabase (service role), `marked` 18, lucide-react, Node 24 (type stripping para tests y seed).

**Spec:** `docs/superpowers/specs/2026-10-13-plan-q4-design.md`

## Global Constraints

- El contenido del plan, las metas, montos y correos **nunca** van al repo (es público). Los tests usan un fixture inventado.
- Toda la interfaz en español con voseo, como el resto del admin ("Escribí", "probá").
- Fechas como `'AAAA-MM-DD'`; "hoy" y las semanas en hora de Costa Rica (UTC−6), semanas de lunes a domingo.
- Archivos en `lib/plan-q4/` que corren en Node (tests, seed) se importan entre sí con extensión `.ts`; los archivos de Next usan `@/lib/...` sin extensión.
- Tokens del admin: fondo `#060607`, superficie `#0c0c0f`, bordes `white/[0.06–0.1]`, acento `#5bb6ff`, rojo `red-300`, verde `emerald-300`, botón principal `bg-[#5bb6ff] text-black hover:bg-[#7cc5ff]`.
- Sin dependencias nuevas.

## Review Focus

- Seed corrido otra vez después de que alguien creó `B8-13` desde el panel y el plan nuevo también trae `B8-13` → no se pisa la del panel, se avisa el choque (`planSeedChanges`, Task 4).

- Títulos con `\|` y `\$` → se guardan como texto plano `|` y `$` (`parsePlan`, Task 2).

```ts
// Fechas del plan en hora de Costa Rica (UTC−6 todo el año, sin horario de verano). Fechas como 'AAAA-MM-DD'.

const CR_OFFSET_MS = 6 * 60 * 60 * 1000
const DAY_MS = 24 * 60 * 60 * 1000
const MESES = ['ene', 'feb', 'mar', 'abr', 'may', 'jun', 'jul', 'ago', 'sep', 'oct', 'nov', 'dic']

export function isValidDate(value: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) return false
  const d = new Date(`${value}T00:00:00Z`)
  return !Number.isNaN(d.getTime()) && d.toISOString().slice(0, 10) === value
}

export function costaRicaToday(now: Date = new Date()): string {
  return new Date(now.getTime() - CR_OFFSET_MS).toISOString().slice(0, 10)
}

export function addDays(date: string, days: number): string {
  return new Date(Date.parse(`${date}T00:00:00Z`) + days * DAY_MS).toISOString().slice(0, 10)
}

// Lunes de la semana de `date`
export function weekStart(date: string): string {
  const dow = new Date(`${date}T00:00:00Z`).getUTCDay() // 0 = domingo
  return addDays(date, dow === 0 ? -6 : 1 - dow)
}

// Domingo de la semana de `date`
export function weekEnd(date: string): string {
  return addDays(weekStart(date), 6)
}

export function isMonday(date: string): boolean {
  return isValidDate(date) && weekStart(date) === date
}

export function daysBetween(from: string, to: string): number {
  return Math.round((Date.parse(`${to}T00:00:00Z`) - Date.parse(`${from}T00:00:00Z`)) / DAY_MS)
}

// '2026-10-16' → '16 oct'
export function formatFecha(date: string): string {
  const [, month, day] = date.split('-').map(Number)
  return `${day} ${MESES[month - 1]}`
}

// Marca de tiempo en hora CR: '13 oct, 14:05'
export function formatStamp(iso: string): string {
  const cr = new Date(Date.parse(iso) - CR_OFFSET_MS).toISOString()
  return `${formatFecha(cr.slice(0, 10))}, ${cr.slice(11, 16)}`
}

export function diasTexto(dias: number): string {
  if (dias === 0) return 'hoy'
  if (dias === 1) return 'mañana'
  return `en ${dias} días`
}
```

```ts
// Parser del archivo del plan (plan-q4-admin.md): bloques, su Contexto en markdown y las tareas del seed.
// Cada bloque: "## B8 — Título · Acción", con "### Tareas (seed)" (tabla) y "### Contexto".
import { isValidDate } from './fechas.ts'
import { PERSONA_NAMES, type Estado, type Hito, type Persona, type ResumenData, type SeedTask, type Tipo } from './types.ts'

export interface PlanBlock {
  key: string // 'b8' o 'decisiones': va en ?tab=
  code: string | null // 'B8'; null en bloques sin clave (Decisiones)
  titulo: string
  tipo: 'lectura' | 'accion'
  contexto: string // markdown
  tareas: SeedTask[]
}

export class PlanParseError extends Error {
  readonly problems: string[]
  constructor(problems: string[]) {
    super(`El plan tiene ${problems.length} error(es):\n- ${problems.join('\n- ')}`)
    this.name = 'PlanParseError'
    this.problems = problems
  }
}

const PART_TWO = /^# Parte 2\b/
const BLOCK_HEADING = /^## (?:(B\d+) — )?(.+?) · (Lectura|Acción)\s*$/
const TABLE_SEPARATOR = /^\|[\s:|-]+\|$/
const PERSONA_BY_NAME = new Map(
  (Object.entries(PERSONA_NAMES) as [Persona, string][]).map(([slug, name]) => [name.toLowerCase(), slug]),
)
const ESTADO_BY_TEXT: Record<string, Estado> = { pendiente: 'pendiente', 'en curso': 'en_curso', hecho: 'hecho' }
const TIPO_BY_TEXT: Record<string, Tipo> = { 'única': 'unica', unica: 'unica', semanal: 'semanal' }
const MESES: Record<string, number> = {
  ene: 1, feb: 2, mar: 3, abr: 4, may: 5, jun: 6, jul: 7, ago: 8, sep: 9, set: 9, oct: 10, nov: 11, dic: 12,
}

const isEmpty = (cell: string) => cell === '—' || cell === ''

function lines(source: string): string[] {
  return source.replace(/\r\n/g, '\n').split('\n')
}

// La Parte 2 del archivo (lo que se guarda en plan_content); el archivo entero si no tiene "# Parte 2"
export function planBody(source: string): string {
  const all = lines(source)
  const partTwo = all.findIndex((line) => PART_TWO.test(line))
  return `${(partTwo >= 0 ? all.slice(partTwo + 1) : all).join('\n').trim()}\n`
}

// Celdas de una fila de tabla markdown; "\|" es una barra dentro de la celda
export function splitRow(line: string): string[] {
  return line.trim().replace(/^\|/, '').replace(/(?<!\\)\|$/, '').split(/(?<!\\)\|/).map((cell) => cell.trim())
}

// Texto plano a partir de markdown escapado: "\$3,500" → "$3,500", "\|" → "|"
export function unescapeMd(text: string): string {
  return text.replace(/\\([\\`*_{}[\]()#+\-.!|$])/g, '$1')
}

function slugify(text: string): string {
  return text.normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')
}

function parseTaskRow(cells: string[], block: { code: string; key: string }, where: string, problems: string[]): SeedTask | null {
  if (cells.length !== 7) {
    problems.push(`${where}: se esperaban 7 columnas (ID, Tarea, Responsable, Apoyo, Fecha, Tipo, Estado) y hay ${cells.length}`)
    return null
  }
  const [id, tituloRaw, responsableRaw, apoyoRaw, fechaRaw, tipoRaw, estadoRaw] = cells
  const before = problems.length

  if (!new RegExp(`^${block.code}-\\d{2,}$`).test(id)) problems.push(`${where}: el id "${id}" no sigue el patrón ${block.code}-NN`)
  const titulo = unescapeMd(tituloRaw)
  if (!titulo) problems.push(`${where}: falta el título`)

  const responsable = PERSONA_BY_NAME.get(responsableRaw.toLowerCase()) ?? null
  if (!responsable) problems.push(`${where}: responsable desconocido "${responsableRaw}"`)
  let apoyo: Persona | null = null
  if (!isEmpty(apoyoRaw)) {
    apoyo = PERSONA_BY_NAME.get(apoyoRaw.toLowerCase()) ?? null
    if (!apoyo) problems.push(`${where}: apoyo desconocido "${apoyoRaw}"`)
    else if (apoyo === responsable) problems.push(`${where}: el apoyo es el mismo responsable`)
  }

  const tipo = TIPO_BY_TEXT[tipoRaw.toLowerCase()] ?? null
  if (!tipo) problems.push(`${where}: tipo desconocido "${tipoRaw}" (única o semanal)`)

  const fechaMatch = /^(desde )?(\d{4}-\d{2}-\d{2})$/.exec(fechaRaw)
  const fecha = fechaMatch && isValidDate(fechaMatch[2]) ? fechaMatch[2] : null
  if (!fecha) problems.push(`${where}: fecha inválida "${fechaRaw}"`)
  else if (tipo && !!fechaMatch?.[1] !== (tipo === 'semanal')) {
    problems.push(`${where}: las semanales llevan "desde AAAA-MM-DD" y las únicas solo la fecha`)
  }

  let estado: Estado | null = null
  if (tipo === 'unica') {
    estado = ESTADO_BY_TEXT[estadoRaw.toLowerCase()] ?? null
    if (!estado) problems.push(`${where}: estado desconocido "${estadoRaw}" (pendiente, en curso o hecho)`)
  } else if (tipo === 'semanal' && !isEmpty(estadoRaw)) {
    problems.push(`${where}: las semanales no llevan estado (usar —)`)
  }

  if (problems.length > before || !responsable || !tipo || !fecha) return null
  return { id, bloque: block.key, titulo, responsable, apoyo, fecha, tipo, estado }
}

export function parsePlan(source: string): PlanBlock[] {
  const problems: string[] = []
  const raw: { heading: string; body: string[] }[] = []
  for (const line of lines(planBody(source))) {
    if (line.startsWith('## ')) raw.push({ heading: line.trim(), body: [] })
    else if (raw.length > 0) raw[raw.length - 1].body.push(line)
    else if (line.trim()) problems.push(`texto antes del primer bloque: "${line.trim().slice(0, 60)}"`)
  }
  if (raw.length === 0) problems.push('no hay ningún bloque ("## B0 — Título · Lectura")')

  const blocks: PlanBlock[] = []
  const seenKeys = new Set<string>()
  const seenIds = new Set<string>()
  for (const { heading, body } of raw) {
    const m = BLOCK_HEADING.exec(heading)
    if (!m) {
      problems.push(`encabezado inválido "${heading}" (formato: "## B8 — Título · Acción")`)
      continue
    }
    const code = m[1] ?? null
    const titulo = m[2].trim()
    const tipo = m[3] === 'Acción' ? 'accion' : 'lectura'
    const key = code ? code.toLowerCase() : slugify(titulo)
    const label = code ?? titulo
    if (seenKeys.has(key)) problems.push(`${label}: bloque repetido`)
    seenKeys.add(key)

    const sections = new Map<string, string[]>()
    let current: string[] | null = null
    for (const line of body) {
      if (line.startsWith('### ')) {
        const name = line.slice(4).trim()
        if (name !== 'Contexto' && name !== 'Tareas (seed)') problems.push(`${label}: sección desconocida "### ${name}"`)
        current = []
        sections.set(name, current)
      } else if (current) current.push(line)
      else if (line.trim()) problems.push(`${label}: texto fuera de "### Contexto"`)
    }

    const contexto = (sections.get('Contexto') ?? []).join('\n').trim()
    if (!contexto) problems.push(`${label}: falta "### Contexto" o está vacío`)

    const tareas: SeedTask[] = []
    const tableLines = sections.get('Tareas (seed)')
    if (tableLines) {
      if (tipo === 'lectura') problems.push(`${label}: un bloque de lectura no lleva tareas`)
      if (!code) problems.push(`${label}: solo los bloques con clave (B0, B1…) llevan tareas`)
      const rows = tableLines.map((l) => l.trim()).filter(Boolean)
      if (rows.some((l) => !l.startsWith('|'))) problems.push(`${label}: hay texto fuera de la tabla de tareas`)
      const tableRows = rows.filter((l) => l.startsWith('|'))
      if (tableRows.length < 2 || !TABLE_SEPARATOR.test(tableRows[1])) {
        problems.push(`${label}: la tabla de tareas necesita encabezado y separador`)
      } else if (code) {
        tableRows.slice(2).forEach((row, i) => {
          const task = parseTaskRow(splitRow(row), { code, key }, `${label} fila ${i + 1}`, problems)
          if (!task) return
          if (seenIds.has(task.id)) problems.push(`${label}: id repetido ${task.id}`)
          seenIds.add(task.id)
          tareas.push(task)
        })
      }
    }
    blocks.push({ key, code, titulo, tipo, contexto, tareas })
  }

  if (problems.length > 0) throw new PlanParseError(problems)
  return blocks
}

// "3 oct arranque · 14 nov revisión" → hitos con fecha
function parseHitos(text: string, year: number): Hito[] {
  const problems: string[] = []
  const hitos = text.split('·').map((part) => part.trim()).filter(Boolean).flatMap((part) => {
    const m = /^(\d{1,2}) (\p{L}{3})\p{L}*\.? (.+)$/u.exec(part)
    const month = m ? MESES[m[2].toLowerCase()] : undefined
    const fecha = m && month ? `${year}-${String(month).padStart(2, '0')}-${m[1].padStart(2, '0')}` : ''
    if (!m || !isValidDate(fecha)) {
      problems.push(`hito sin fecha legible: "${part}" (formato: "3 oct texto")`)
      return []
    }
    return [{ fecha, titulo: m[3].charAt(0).toUpperCase() + m[3].slice(1) }]
  })
  if (problems.length > 0) throw new PlanParseError(problems)
  return hitos.sort((a, b) => a.fecha.localeCompare(b.fecha))
}

// Encabezado fijo del Resumen: viñetas "- **Etiqueta:** valor" bajo "### Pestaña Resumen" (Parte 1).
// null si el archivo no trae esa sección.
export function parseResumen(source: string): ResumenData | null {
  const all = lines(source)
  const start = all.findIndex((line) => /^### Pestaña Resumen\s*$/.test(line))
  if (start < 0) return null
  const year = Number(/Q4 (\d{4})/.exec(source)?.[1] ?? new Date().getUTCFullYear())
  const resumen: ResumenData = { encabezado: [], hitos: [] }
  for (const line of all.slice(start + 1)) {
    if (/^#{1,3} /.test(line)) break
    const m = /^- \*\*(.+?):\*\*\s*(.+)$/.exec(line.trim())
    if (!m) continue
    const label = unescapeMd(m[1])
    const value = unescapeMd(m[2].trim())
    if (label.toLowerCase() === 'hitos') resumen.hitos = parseHitos(value, year)
    else resumen.encabezado.push({ label, value })
  }
  return resumen.encabezado.length > 0 || resumen.hitos.length > 0 ? resumen : null
}
```

```ts
// Reglas del Plan Q4: vencidas, semana actual, semanales, pendientes, orden, resumen, ids nuevos y cambios.
import { daysBetween, weekEnd, weekStart } from './fechas.ts'
import { PERSONAS, isPersona, type Estado, type Hito, type Persona, type PlanTask } from './types.ts'

export type PersonaFilter = Persona | 'todos'
export const RESUMEN_TAB = 'resumen'

type Fechada = Pick<PlanTask, 'tipo' | 'estado' | 'fecha'>

export function isOverdue(task: Fechada, today: string): boolean {
  return task.tipo === 'unica' && task.estado !== 'hecho' && task.fecha < today
}

export function isDueThisWeek(task: Fechada, today: string): boolean {
  return task.tipo === 'unica' && task.fecha >= weekStart(today) && task.fecha <= weekEnd(today)
}

// Una semanal aparece todas las semanas desde la de su fecha de inicio
export function isActiveWeekly(task: Fechada, today: string): boolean {
  return task.tipo === 'semanal' && task.fecha <= weekEnd(today)
}

// doneThisWeek: ids de las semanales con check en la semana actual
export function isPending(task: Fechada & Pick<PlanTask, 'id'>, today: string, doneThisWeek: ReadonlySet<string>): boolean {
  if (task.tipo === 'unica') return task.estado !== 'hecho'
  return isActiveWeekly(task, today) && !doneThisWeek.has(task.id)
}

export function matchesPersona(task: Pick<PlanTask, 'responsable'>, persona: PersonaFilter): boolean {
  return persona === 'todos' || task.responsable === persona
}

// Únicas abiertas por fecha (las vencidas quedan primero), después las semanales y al final las hechas
export function compareTasks(a: PlanTask, b: PlanTask): number {
  const rank = (t: PlanTask) => (t.tipo === 'semanal' ? 1 : t.estado === 'hecho' ? 2 : 0)
  return rank(a) - rank(b) || a.fecha.localeCompare(b.fecha) || a.id.localeCompare(b.id, 'en', { numeric: true })
}

export function pendingByBlock(
  tasks: readonly PlanTask[],
  today: string,
  doneThisWeek: ReadonlySet<string>,
  persona: PersonaFilter,
): Map<string, number> {
  const counts = new Map<string, number>()
  for (const t of tasks) {
    if (matchesPersona(t, persona) && isPending(t, today, doneThisWeek)) counts.set(t.bloque, (counts.get(t.bloque) ?? 0) + 1)
  }
  return counts
}

export interface PersonWeek {
  persona: Persona
  vencidas: PlanTask[]
  semana: PlanTask[]
  semanales: PlanTask[]
}

// "Esta semana" del Resumen, por responsable. Una vencida no se repite en la semana.
export function weekSummary(tasks: readonly PlanTask[], today: string, persona: PersonaFilter): PersonWeek[] {
  return PERSONAS.filter((p) => persona === 'todos' || p === persona).map((p) => {
    const mine = tasks.filter((t) => t.responsable === p).sort(compareTasks)
    return {
      persona: p,
      vencidas: mine.filter((t) => isOverdue(t, today)),
      semana: mine.filter((t) => isDueThisWeek(t, today) && !isOverdue(t, today)),
      semanales: mine.filter((t) => isActiveWeekly(t, today)),
    }
  })
}

// Avance de las únicas por bloque
export function blockProgress(tasks: readonly PlanTask[], persona: PersonaFilter): Map<string, { hechas: number; total: number }> {
  const progress = new Map<string, { hechas: number; total: number }>()
  for (const t of tasks) {
    if (t.tipo !== 'unica' || !matchesPersona(t, persona)) continue
    const p = progress.get(t.bloque) ?? { hechas: 0, total: 0 }
    p.total += 1
    if (t.estado === 'hecho') p.hechas += 1
    progress.set(t.bloque, p)
  }
  return progress
}

export function nextMilestone(hitos: readonly Hito[], today: string): { hito: Hito; dias: number } | null {
  const hito = [...hitos].sort((a, b) => a.fecha.localeCompare(b.fecha)).find((h) => h.fecha >= today)
  return hito ? { hito, dias: daysBetween(today, hito.fecha) } : null
}

export function nextTaskId(code: string, existingIds: readonly string[]): string {
  const prefix = `${code}-`
  const max = existingIds
    .filter((id) => id.startsWith(prefix))
    .map((id) => Number(id.slice(prefix.length)))
    .filter(Number.isInteger)
    .reduce((m, n) => Math.max(m, n), 0)
  return `${prefix}${String(max + 1).padStart(2, '0')}`
}

// Pestaña y persona de la URL; lo inválido cae en Resumen y en la persona del usuario
export function resolveTab(raw: string | undefined, keys: readonly string[]): string {
  return raw && keys.includes(raw) ? raw : RESUMEN_TAB
}

export function resolvePersona(raw: string | undefined, actor: Persona | null): PersonaFilter {
  if (raw === 'todos') return 'todos'
  if (isPersona(raw)) return raw
  return actor ?? 'todos'
}

export function planHref(tab: string, persona: PersonaFilter): string {
  return `/plan-q4?tab=${encodeURIComponent(tab)}&persona=${persona}`
}

export interface TaskPatch {
  estado?: Estado

  nota?: string | null

  nota: string

// Solo lo que cambió en el formulario de edición. Delegar a quien era el apoyo deja la tarea sin apoyo.
export function diffTask(task: PlanTask, form: TaskForm): TaskPatch {
  const patch: TaskPatch = {}
  if (task.tipo === 'unica' && form.estado !== task.estado) patch.estado = form.estado
  if (form.responsable !== task.responsable) patch.responsable = form.responsable
  const apoyo = form.apoyo === form.responsable ? null : form.apoyo
  if (apoyo !== task.apoyo) patch.apoyo = apoyo
  if (form.fecha !== task.fecha) patch.fecha = form.fecha
  const nota = form.nota.trim() || null
  if (nota !== task.nota) patch.nota = nota
  return patch
}

// Cambios a guardar en la base, firmados por quien los hizo
export function buildUpdate(
  current: Pick<PlanTask, 'tipo' | 'responsable' | 'apoyo'>,
  patch: TaskPatch,
  actor: Persona,
  now: string,
): { ok: true; update: Record<string, unknown> } | { ok: false; error: string } {
  if (patch.estado !== undefined && current.tipo === 'semanal') {
    return { ok: false, error: 'Las semanales no tienen estado: se marcan por semana' }
  }
  const update: Record<string, unknown> = { ...patch, actualizado_por: actor, actualizado_en: now }
  const responsable = patch.responsable ?? current.responsable
  const apoyo = patch.apoyo !== undefined ? patch.apoyo : current.apoyo
  if (apoyo !== null && apoyo === responsable) update.apoyo = null
  return { ok: true, update }
}

// Un check semanal solo vale para semanales y desde la semana en que arrancan
export function weekToggleError(task: Pick<PlanTask, 'tipo' | 'fecha'>, week: string): string | null {
  if (task.tipo !== 'semanal') return 'Solo las semanales se marcan por semana'
  if (week < weekStart(task.fecha)) return 'Esa semana es anterior al inicio de la tarea'
  return null
}
```

```ts
// Validación de lo que llega a /api/plan-q4 (lógica pura): el valor limpio o un error en español.
import { isMonday, isValidDate } from './fechas.ts'
import { isEstado, isPersona, type Persona, type Tipo } from './types.ts'
import type { TaskPatch } from './rules.ts'

export type Parsed<T> = { ok: true; value: T } | { ok: false; error: string }

export const MAX_TITULO = 300
export const MAX_NOTA = 2000

export interface NewTask {
  bloque: string
  titulo: string
  responsable: Persona
  apoyo: Persona | null
  fecha: string
  tipo: Tipo
}

const fail = (error: string): { ok: false; error: string } => ({ ok: false, error })

function asObject(body: unknown): Record<string, unknown> | null {
  return body && typeof body === 'object' && !Array.isArray(body) ? (body as Record<string, unknown>) : null
}

export function parseTaskPatch(body: unknown): Parsed<TaskPatch> {
  const o = asObject(body)
  if (!o) return fail('Datos inválidos')
  const value: TaskPatch = {}
  if ('estado' in o) {
    if (!isEstado(o.estado)) return fail('Estado inválido')
    value.estado = o.estado
  }
  if ('responsable' in o) {
    if (!isPersona(o.responsable)) return fail('Responsable inválido')
    value.responsable = o.responsable
  }
  if ('apoyo' in o) {
    const apoyo = o.apoyo
    if (apoyo === null) value.apoyo = null
    else if (isPersona(apoyo)) value.apoyo = apoyo
    else return fail('Apoyo inválido')
  }
  if ('fecha' in o) {
    if (typeof o.fecha !== 'string' || !isValidDate(o.fecha)) return fail('Fecha inválida')
    value.fecha = o.fecha
  }
  if ('nota' in o) {
    if (o.nota !== null && typeof o.nota !== 'string') return fail('Nota inválida')
    const nota = typeof o.nota === 'string' ? o.nota.trim() : ''
    if (nota.length > MAX_NOTA) return fail(`La nota pasa de ${MAX_NOTA} caracteres`)
    value.nota = nota || null
  }
  if (Object.keys(value).length === 0) return fail('No hay cambios')
  return { ok: true, value }
}

export function parseNewTask(body: unknown, bloques: readonly string[]): Parsed<NewTask> {
  const o = asObject(body)
  if (!o) return fail('Datos inválidos')
  const { bloque, responsable, fecha, tipo } = o
  if (typeof bloque !== 'string' || !bloques.includes(bloque)) return fail('Bloque inválido')
  const titulo = typeof o.titulo === 'string' ? o.titulo.trim() : ''
  if (!titulo) return fail('Escribí el título de la tarea')
  if (titulo.length > MAX_TITULO) return fail(`El título pasa de ${MAX_TITULO} caracteres`)
  if (!isPersona(responsable)) return fail('Responsable inválido')
  let apoyo: Persona | null = null
  if (o.apoyo !== undefined && o.apoyo !== null && o.apoyo !== '') {
    if (!isPersona(o.apoyo)) return fail('Apoyo inválido')
    apoyo = o.apoyo === responsable ? null : o.apoyo
  }
  if (typeof fecha !== 'string' || !isValidDate(fecha)) return fail('Fecha inválida')
  if (tipo !== 'unica' && tipo !== 'semanal') return fail('Tipo inválido')
  return { ok: true, value: { bloque, titulo, responsable, apoyo, fecha, tipo } }
}

export function parseWeekToggle(body: unknown): Parsed<{ week_start: string; done: boolean }> {
  const o = asObject(body)
  if (!o) return fail('Datos inválidos')
  if (typeof o.week_start !== 'string' || !isMonday(o.week_start)) return fail('La semana tiene que empezar en lunes')
  if (typeof o.done !== 'boolean') return fail('Falta indicar si quedó hecha')
  return { ok: true, value: { week_start: o.week_start, done: o.done } }
}
```

```ts
// Qué hace el seed con las tareas del archivo frente a las de la base, y el equipo del plan (lógica pura).
import { isPersona, type Origen, type Persona, type SeedTask } from './types.ts'

export const PLAN_TEAM: { slug: Persona; rol: string }[] = [
  { slug: 'ale', rol: 'Fundador: decide y aprueba' },
  { slug: 'soshil', rol: 'Parte técnica: desarrollo, guiones y edición' },
  { slug: 'alina', rol: 'Closer: llamadas, leads y pipeline' },
]

export interface ExistingTask {
  id: string
  titulo: string
  origen: Origen
}

export interface SeedChanges {
  nuevas: SeedTask[] // se insertan completas
  renombradas: SeedTask[] // solo cambia el título
  choques: string[] // el id ya lo usa una tarea creada desde el panel: no se toca
  soloEnBase: string[] // no se borran
  sinCambios: number
}

export function planSeedChanges(fileTasks: readonly SeedTask[], existing: readonly ExistingTask[]): SeedChanges {
  const byId = new Map(existing.map((t) => [t.id, t]))
  const changes: SeedChanges = { nuevas: [], renombradas: [], choques: [], soloEnBase: [], sinCambios: 0 }
  for (const t of fileTasks) {
    const current = byId.get(t.id)
    if (!current) changes.nuevas.push(t)
    else if (current.origen === 'panel') changes.choques.push(t.id)
    else if (current.titulo !== t.titulo) changes.renombradas.push(t)
    else changes.sinCambios += 1
  }
  const inFile = new Set(fileTasks.map((t) => t.id))
  changes.soloEnBase = existing.filter((t) => !inFile.has(t.id)).map((t) => t.id)
  return changes
}

// PLAN_Q4_EMAILS="ale:correo,soshil:correo" → { ale: 'correo', soshil: 'correo' }
export function parseTeamEmails(raw: string | undefined): Partial<Record<Persona, string>> {
  const emails: Partial<Record<Persona, string>> = {}
  for (const pair of (raw ?? '').split(',')) {
    const [slug, email] = pair.split(':').map((s) => s.trim())
    if (isPersona(slug) && email?.includes('@')) emails[slug] = email.toLowerCase()
  }
  return emails
}
```

```ts
import { createServiceClient } from '@/lib/supabase/service'

  userId: string
  slug: Persona | null // null: tiene acceso pero no es del equipo del plan (solo lee)

export const getPlanActor = cache(async (): Promise<PlanActor | null> => {
  const { user } = await getCurrentSession()
  if (!user) return null
  const { data } = await createServiceClient().from('plan_team').select('slug').eq('user_id', user.id).maybeSingle()
  return { userId: user.id, slug: (data?.slug as Persona | undefined) ?? null }
})

export interface PlanContent {
  blocks: PlanBlock[]
  resumen: ResumenData
  actualizadoEn: string
}

// null si el seed todavía no corrió
export const getPlanContent = cache(async (): Promise<PlanContent | null> => {
  const { data, error } = await createServiceClient()
    .from('plan_content')
    .select('markdown, resumen, actualizado_en')
    .eq('id', 1)
    .maybeSingle()
  if (error) throw error
  if (!data) return null
  return { blocks: parsePlan(data.markdown), resumen: data.resumen as ResumenData, actualizadoEn: data.actualizado_en }
})

const TASK_COLUMNS = 'id, bloque, titulo, responsable, apoyo, fecha, tipo, estado, nota, origen, actualizado_por, actualizado_en'

export async function getPlanTasks(week: string): Promise<{ tasks: PlanTask[]; checks: WeekCheck[] }> {
  const service = createServiceClient()
  const [tasks, checks] = await Promise.all([
    service.from('plan_tasks').select(TASK_COLUMNS),
    service.from('plan_task_weeks').select('task_id, week_start, done, done_by, done_at').eq('week_start', week),
  ])
  if (tasks.error) throw tasks.error
  if (checks.error) throw checks.error
  return { tasks: tasks.data as PlanTask[], checks: checks.data as WeekCheck[] }
}
```

```ts
import { NextResponse } from 'next/server'
import { createServiceClient } from '@/lib/supabase/service'
import { jsonError, readJson, requirePlanMember } from '@/lib/plan-q4/api'
import { getPlanContent } from '@/lib/plan-q4/data'
import { nextTaskId } from '@/lib/plan-q4/rules'
import { parseNewTask } from '@/lib/plan-q4/validate'

export async function POST(request: Request) {
  const { slug, denied } = await requirePlanMember()
  if (denied) return denied
  const content = await getPlanContent()
  if (!content) return jsonError('El plan todavía no está cargado', 409)
  const actionBlocks = content.blocks.filter((b) => b.tipo === 'accion' && b.code)
  const parsed = parseNewTask(await readJson(request), actionBlocks.map((b) => b.key))
  if (!parsed.ok) return jsonError(parsed.error, 400)
  const code = actionBlocks.find((b) => b.key === parsed.value.bloque)!.code!

  const service = createServiceClient()
  // Dos intentos: si alguien creó otra tarea en el mismo bloque al mismo tiempo, choca la PK
  for (let attempt = 0; attempt < 2; attempt++) {
    const { data: rows, error: idsError } = await service.from('plan_tasks').select('id').eq('bloque', parsed.value.bloque)
    if (idsError) return jsonError(idsError.message, 500)
    const id = nextTaskId(code, (rows ?? []).map((r) => r.id as string))
    const { error } = await service.from('plan_tasks').insert({
      id,
      ...parsed.value,
      estado: parsed.value.tipo === 'unica' ? 'pendiente' : null,
      origen: 'panel',
      actualizado_por: slug,
      actualizado_en: new Date().toISOString(),
    })
    if (!error) return NextResponse.json({ id }, { status: 201 })
    if (error.code !== '23505') return jsonError(error.message, 500)
  }
  return jsonError('No se pudo asignar un número a la tarea, probá de nuevo', 409)
}
```

```ts
import { NextResponse } from 'next/server'
import { createServiceClient } from '@/lib/supabase/service'
import { jsonError, readJson, requirePlanMember } from '@/lib/plan-q4/api'
import { buildUpdate } from '@/lib/plan-q4/rules'
import type { PlanTask } from '@/lib/plan-q4/types'
import { parseTaskPatch } from '@/lib/plan-q4/validate'

export async function PATCH(request: Request, { params }: { params: Promise<{ id: string }> }) {
  const { slug, denied } = await requirePlanMember()
  if (denied) return denied
  const { id } = await params
  const parsed = parseTaskPatch(await readJson(request))
  if (!parsed.ok) return jsonError(parsed.error, 400)

  const service = createServiceClient()
  const { data: current, error: readError } = await service
    .from('plan_tasks')
    .select('tipo, responsable, apoyo')
    .eq('id', id)
    .maybeSingle()
  if (readError) return jsonError(readError.message, 500)
  if (!current) return jsonError('Tarea no encontrada', 404)

  const result = buildUpdate(current as Pick<PlanTask, 'tipo' | 'responsable' | 'apoyo'>, parsed.value, slug, new Date().toISOString())
  if (!result.ok) return jsonError(result.error, 400)
  const { error } = await service.from('plan_tasks').update(result.update).eq('id', id)
  if (error) return jsonError(error.message, 500)
  return NextResponse.json({ ok: true })
}
```

```ts
import { NextResponse } from 'next/server'
import { createServiceClient } from '@/lib/supabase/service'
import { jsonError, readJson, requirePlanMember } from '@/lib/plan-q4/api'
import { weekToggleError } from '@/lib/plan-q4/rules'
import type { PlanTask } from '@/lib/plan-q4/types'
import { parseWeekToggle } from '@/lib/plan-q4/validate'

// Check de una semanal en una semana (lunes)
export async function PUT(request: Request, { params }: { params: Promise<{ id: string }> }) {
  const { slug, denied } = await requirePlanMember()
  if (denied) return denied
  const { id } = await params
  const parsed = parseWeekToggle(await readJson(request))
  if (!parsed.ok) return jsonError(parsed.error, 400)

  const service = createServiceClient()
  const { data: task, error: readError } = await service.from('plan_tasks').select('tipo, fecha').eq('id', id).maybeSingle()
  if (readError) return jsonError(readError.message, 500)
  if (!task) return jsonError('Tarea no encontrada', 404)
  const invalid = weekToggleError(task as Pick<PlanTask, 'tipo' | 'fecha'>, parsed.value.week_start)
  if (invalid) return jsonError(invalid, 400)

  const now = new Date().toISOString()
  const { done, week_start } = parsed.value
  const { error } = await service.from('plan_task_weeks').upsert(
    { task_id: id, week_start, done, done_by: done ? slug : null, done_at: done ? now : null },
    { onConflict: 'task_id,week_start' },
  )
  if (error) return jsonError(error.message, 500)
  const { error: stampError } = await service.from('plan_tasks').update({ actualizado_por: slug, actualizado_en: now }).eq('id', id)
  if (stampError) return jsonError(stampError.message, 500)
  return NextResponse.json({ ok: true })
}
```

```ts
// Envía JSON a las APIs del plan; devuelve el mensaje de error, o null si salió bien
export async function sendJson(url: string, method: 'POST' | 'PATCH' | 'PUT', body: unknown): Promise<string | null> {
  try {
    const res = await fetch(url, { method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
    if (res.ok) return null
    const data: unknown = await res.json().catch(() => null)
    const error = data && typeof data === 'object' && 'error' in data ? (data as { error: unknown }).error : null
    return typeof error === 'string' ? error : 'No se pudo guardar'
  } catch {
    return 'Sin conexión, probá de nuevo'
  }
}
```

```tsx
import { useRouter } from 'next/navigation'
import { Check, ChevronDown, Loader2, Repeat, StickyNote } from 'lucide-react'

export function TaskItem({ task, today, week, doneThisWeek, canEdit }: Props) {
  const router = useRouter()
  const [open, setOpen] = useState(false)
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const [refreshing, startRefresh] = useTransition()
  const working = busy || refreshing
  const semanal = task.tipo === 'semanal'

  const overdue = isOverdue(task, today)
  const thisWeek = !overdue && !done && isDueThisWeek(task, today)
  const url = `/api/plan-q4/tasks/${encodeURIComponent(task.id)}`

  async function save(send: () => Promise<string | null>): Promise<boolean> {
    setBusy(true)
    setError(null)
    const err = await send()
    setBusy(false)
    if (err) {
      setError(err)
      return false
    }
    startRefresh(() => router.refresh())
    return true
  }

  function toggleDone() {
    void save(() =>
      semanal
        ? sendJson(`${url}/weeks`, 'PUT', { week_start: week, done: !done })
        : sendJson(url, 'PATCH', { estado: done ? 'pendiente' : 'hecho' }),

interface EditorProps {
  task: PlanTask
  canEdit: boolean
  working: boolean
  error: string | null
  onCancel: () => void
  onSave: (patch: object) => void
}

function TaskEditor({ task, canEdit, working, error, onCancel, onSave }: EditorProps) {
  const [estado, setEstado] = useState<Estado>(task.estado ?? 'pendiente')
  const [responsable, setResponsable] = useState<Persona>(task.responsable)
  const [apoyo, setApoyo] = useState<Persona | ''>(task.apoyo ?? '')
  const [fecha, setFecha] = useState(task.fecha)
  const [nota, setNota] = useState(task.nota ?? '')

  function submit(e: React.FormEvent) {
    e.preventDefault()
    const patch = diffTask(task, { estado, responsable, apoyo: apoyo || null, fecha, nota })
    if (Object.keys(patch).length === 0) onCancel()
    else onSave(patch)
  }

  return (
    <form onSubmit={submit} className="mt-3 sm:ml-8 grid gap-3 sm:grid-cols-2">
      {task.tipo === 'unica' && (
        <Field label="Estado">
          <select value={estado} onChange={(e) => setEstado(e.target.value as Estado)} disabled={!canEdit} className={FIELD_CLASS}>
            {ESTADOS.map((s) => (
              <option key={s} value={s}>{ESTADO_LABELS[s]}</option>
            ))}
          </select>
        </Field>
      )}
```

```tsx
  return (
    <div className="space-y-8">
```
